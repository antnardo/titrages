% nom, grandeur, argument, valeur
\cs_new_protected:Npn \ligne #1 #2 #3
  { \iow_now:Nx \g_test_iow { #1 ~ #2 ~ #3 ~ \use:c { titrage #2 } { #1 } { #3 } } }

  { \clist_map_inline:nn { #3 } { \ligne { #1 } { #2 } { ##1 } } }
% sous pgf : l'expression tracée, évaluée par pgfmath comme dans l'axe
\str_if_eq:VnT \g__titrages_moteur_str { pgf }
  {
    \fp_new:N \l_x_fp
    \pgfmathdeclarefunction{x}{0}{ \pgfmathparse { \fp_use:N \l_x_fp } }
    \cs_new_protected:Npn \pgfmathligne #1 #2
      {
        \fp_set:Nn \l_x_fp { #2 }
        \group_begin:
          \pgfkeys { /pgf/fpu = true , /pgf/fpu/output~format = fixed }
          \pgfmathparse { \tl_use:c { g__titrages_ #1 _V_tl } }
          \iow_now:Nx \g_test_iow { #1 ~ pgfmath ~ #2 ~ \pgfmathresult }
        \group_end:
      }
    \cs_new_protected:Npn \pgfmathlignes #1 #2 { \clist_map_inline:nn { #2 } { \pgfmathligne { #1 } { ##1 } } }
  }
